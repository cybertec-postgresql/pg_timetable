    54  // Execute implements the svc.Handler interface and serves control requests
    55  // from the Service Control Manager until the service is stopped.
    56  func (winServiceRunner) Execute(_ []string, request <-chan svc.ChangeRequest, status chan<- svc.Status) (bool, uint32) {
    57  	notify.SetGlobalStatus(status)
    58  	status <- svc.Status{State: svc.StartPending}
    59  	// Report Running with the controls we accept so the SCM stops showing the
    60  	// service as "Starting" and offers a working Stop command. Without this the
    61  	// service is stuck in StartPending and svc.Stop fails with
    62  	// ERROR_INVALID_SERVICE_CONTROL.
    63  	status <- svc.Status{State: svc.Running, Accepts: svc.AcceptStop | svc.AcceptShutdown}
    64  	for req := range request {
    65  		switch req.Cmd {
    66  		case svc.Interrogate:
    67  			status <- req.CurrentStatus
    68  		case svc.Stop, svc.Shutdown:
    69  			status <- svc.Status{State: svc.StopPending}
    70  			cancelApplication(status)
    71  			return false, 0
    72  		}
    73  	}
    74  	return false, 0
    75  }

    77  // cancelApplication triggers a graceful shutdown of the running daemon. The
    78  // Service Control Manager may request a stop while main is still starting up
    79  // and has not assigned the cancellation function yet, so wait until it is
    80  // ready (bounded by serviceStopTimeout), reporting StopPending heartbeats so
    81  // the SCM does not consider the service hung.
    82  func cancelApplication(status chan<- svc.Status) {
    83  	deadline := time.After(serviceStopTimeout)
    84  	ticker := time.NewTicker(servicePollingInterval)
    85  	defer ticker.Stop()
    86  	for {
    87  		if cancel := getCancelFn(); cancel != nil {
    88  			cancel()
    89  			return
    90  		}
    91  		select {
    92  		case <-cancelReady:
    93  			if cancel := getCancelFn(); cancel != nil {
    94  				cancel()
    95  			}
    96  			return
    97  		case <-deadline:
    98  			// Give up waiting; the process will be terminated by the SCM.
    99  			return
   100  		case <-ticker.C:
   101  			status <- svc.Status{State: svc.StopPending}
   102  		}
   103  	}
   104  }

   133  // serviceInstall registers the executable as a Windows service. The command
   134  // line arguments are persisted so that the service runs the daemon with the
   135  // same connection settings used during installation.
   136  func serviceInstall(name string, cmdOpts *config.CmdOptions, exePath string) int {
   137  	if err := validateServiceAccount(cmdOpts.Service.ServiceUser, cmdOpts.Service.ServicePassword); err != nil {
   138  		fmt.Println(err)
   139  		return ExitCodeFatalError
   140  	}
   141  	m, err := connectManager()
   142  	if err != nil {
   143  		fmt.Println(err)
   144  		return ExitCodeFatalError
   145  	}
   146  	defer func() { _ = m.Disconnect() }()
   147  
   148  	if s, err := m.OpenService(name); err == nil {
   149  		s.Close()
   150  		fmt.Printf("Service %q already exists\n", name)
   151  		return ExitCodeFatalError
   152  	}
   153  	args := filterServiceArgs(os.Args[1:])
   154  	for _, warning := range relativePathWarnings(args) {
   155  		fmt.Println("Warning:", warning)
   156  	}
   157  	s, err := m.CreateService(name, exePath, newServiceConfig(name, cmdOpts), args...)
   158  	if err != nil {
   159  		fmt.Println("Failed to install service:", err)
   160  		return ExitCodeFatalError
   161  	}
   162  	defer s.Close()
   163  	actions := []mgr.RecoveryAction{{Type: mgr.ServiceRestart, Delay: serviceRestartDelay}}
   164  	if err := s.SetRecoveryActions(actions, serviceRecoveryReset); err != nil {
   165  		fmt.Println("Service installed, but recovery actions were not configured:", err)
   166  	}
   167  	fmt.Printf("Service %q installed successfully\n", name)
   168  	return ExitCodeOK
   169  }

   220  // serviceControl handles the start, stop, restart, and status actions.
   221  func serviceControl(name, action string) int {
   222  	m, err := connectManager()
   223  	if err != nil {
   224  		fmt.Println(err)
   225  		return ExitCodeFatalError
   226  	}
   227  	defer func() { _ = m.Disconnect() }()
   228  
   229  	s, err := m.OpenService(name)
   230  	if err != nil {
   231  		fmt.Printf("Service %q is not installed\n", name)
   232  		return ExitCodeFatalError
   233  	}
   234  	defer s.Close()
   235  
   236  	switch action {
   237  	case "start":
   238  		err = s.Start()
   239  		if err == nil {
   240  			fmt.Printf("Service %q started\n", name)
   241  		}
   242  	case "stop":
   243  		err = stopService(s)
   244  		if err == nil {
   245  			fmt.Printf("Service %q stopped\n", name)
   246  		}
   247  	case "restart":
   248  		if err = stopService(s); err == nil {
   249  			err = s.Start()
   250  		}
   251  		if err == nil {
   252  			fmt.Printf("Service %q restarted\n", name)
   253  		}
   254  	case "status":
   255  		var status svc.Status
   256  		if status, err = s.Query(); err == nil {
   257  			fmt.Printf("Service %q: %s\n", name, serviceStateString(status.State))
   258  		}
   259  	}
   260  	if err != nil {
   261  		fmt.Printf("Failed to %s service: %v\n", action, err)
   262  		return ExitCodeFatalError
   263  	}
   264  	return ExitCodeOK
   265  }

   267  // stopService requests the service to stop and waits until it reports the
   268  // stopped state or the timeout expires. A service that is already stopped is
   269  // reported by the SCM as ERROR_SERVICE_NOT_ACTIVE, which is treated as success
   270  // so that stop, restart, and uninstall work on an installed but idle service.
   271  func stopService(s *mgr.Service) error {
   272  	status, err := s.Control(svc.Stop)
   273  	if err != nil {
   274  		// A service that is already stopped reports ERROR_SERVICE_NOT_ACTIVE.
   275  		// A service stuck in StartPending (or otherwise unable to accept a
   276  		// stop) reports ERROR_INVALID_SERVICE_CONTROL. Neither should block
   277  		// removal, so treat both as "nothing left to stop".
   278  		if errors.Is(err, windows.ERROR_SERVICE_NOT_ACTIVE) ||
   279  			errors.Is(err, windows.ERROR_INVALID_SERVICE_CONTROL) {
   280  			return nil
   281  		}
   282  		return err
   283  	}
   284  	if status.State == svc.Stopped {
   285  		return nil
   286  	}
   287  	for deadline := time.Now().Add(serviceStopTimeout); time.Now().Before(deadline); <-time.After(servicePollingInterval) {
   288  		if status, err = s.Query(); err != nil {
   289  			return err
   290  		}
   291  		if status.State == svc.Stopped {
   292  			return nil
   293  		}
   294  	}
   295  	return fmt.Errorf("timed out waiting for service to stop")
   296  }

   303  	case svc.StartPending:

   317  	}

   320  // filterServiceArgs removes the service management flags from the argument
   321  // list so they are not persisted into the installed service command line.
   322  // This also keeps the service account credentials out of the registry.
   323  func filterServiceArgs(args []string) []string {
   324  	filtered := make([]string, 0, len(args))
   325  	for i := 0; i < len(args); i++ {
   326  		arg := args[i]
   327  		name := strings.TrimLeft(arg, "-")
   328  		if eq := strings.IndexByte(name, '='); eq >= 0 {
   329  			name = name[:eq]
   330  		}
   331  		switch name {
   332  		case "service", "service-name", "service-user", "service-password":
   333  			if !strings.Contains(arg, "=") {
   334  				i++ // skip the option value too
   335  			}
   336  			continue
   337  		}
   338  		filtered = append(filtered, arg)
   339  	}
   340  	return filtered
   341  }

   351  // relativePathWarnings returns warnings for path arguments that would be
   352  // resolved against the executable directory when running as a service.
   353  func relativePathWarnings(args []string) []string {
   354  	var warnings []string
   355  	for i := 0; i < len(args); i++ {
   356  		arg := args[i]
   357  		name, value := strings.TrimLeft(arg, "-"), ""
   358  		hasValue := false
   359  		if before, after, ok := strings.Cut(arg, "="); ok {
   360  			name, value, hasValue = strings.TrimLeft(before, "-"), after, true
   361  		} else if filePathFlags[name] && i+1 < len(args) && !strings.HasPrefix(args[i+1], "-") {
   362  			value, hasValue = args[i+1], true
   363  			i++
   364  		}
   365  		if hasValue && filePathFlags[name] && !filepath.IsAbs(value) {
   366  			warnings = append(warnings, fmt.Sprintf(
   367  				"path %q given for --%s is relative and will be resolved against the executable directory when running as a service",
   368  				value, name))
   369  		}
   370  	}
   371  	return warnings
   372  }
