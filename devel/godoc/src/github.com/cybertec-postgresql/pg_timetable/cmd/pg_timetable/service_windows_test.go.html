   149  // SCM channels: it must report StartPending, answer interrogation, and on a
   150  // stop request report StopPending and trigger the application cancellation.
   151  func TestWinServiceRunnerExecute(t *testing.T) {
   152  	requests := make(chan svc.ChangeRequest)
   153  	statuses := make(chan svc.Status, 8)
   154  	cancelled := make(chan struct{})
   155  
   156  	prev := saveCancelState()
   157  	setCancelFn(func() { close(cancelled) })
   158  	defer func() {
   159  		prev.restore()
   160  		notify.SetGlobalStatus(nil)
   161  	}()
   162  
   163  	result := make(chan [2]uint32)
   164  	go func() {
   165  		shutdown, code := (winServiceRunner{}).Execute(nil, requests, statuses)
   166  		result <- [2]uint32{boolToUint32(shutdown), code}
   167  	}()
   168  
   169  	select {
   170  	case st := <-statuses:
   171  		assert.Equal(t, svc.StartPending, st.State)
   172  	case <-time.After(time.Second):
   173  		t.Fatal("no initial status reported")
   174  	}
   175  
   176  	// Immediately after StartPending the handler must report Running and
   177  	// advertise that it accepts Stop and Shutdown, otherwise the SCM keeps the
   178  	// service in the Starting state and refuses control requests.
   179  	select {
   180  	case st := <-statuses:
   181  		assert.Equal(t, svc.Running, st.State)
   182  		assert.Equal(t, svc.AcceptStop|svc.AcceptShutdown, st.Accepts)
   183  	case <-time.After(time.Second):
   184  		t.Fatal("no running status reported")
   185  	}
   186  
   187  	requests <- svc.ChangeRequest{Cmd: svc.Interrogate, CurrentStatus: svc.Status{State: svc.Running}}
   188  	select {
   189  	case st := <-statuses:
   190  		assert.Equal(t, svc.Running, st.State)
   191  	case <-time.After(time.Second):
   192  		t.Fatal("interrogate request was not answered")
   193  	}
   194  
   195  	requests <- svc.ChangeRequest{Cmd: svc.Stop}
   196  	select {
   197  	case st := <-statuses:
   198  		assert.Equal(t, svc.StopPending, st.State)
   199  	case <-time.After(time.Second):
   200  		t.Fatal("no stop pending status reported")
   201  	}
   202  
   203  	res := <-result
   204  	assert.Equal(t, uint32(0), res[0], "handler must not request process shutdown")
   205  	assert.Equal(t, uint32(0), res[1], "exit code must be clean")
   206  	select {
   207  	case <-cancelled:
   208  	case <-time.After(time.Second):
   209  		t.Fatal("cancellation function was not called")
   210  	}
   211  }

   363  // TestServiceRealStartStop answers "can we start a real application in tests?"
   364  // with yes: it installs the test binary as a service and actually starts it.
   365  // The test binary is svc.Run-aware because it shares this package's init(),
   366  // which launches winServiceRunner when the SCM starts the process. TestMain
   367  // (below) detects that service launch, installs a cancel function so Stop is
   368  // instant, and blocks so the test suite itself does not run in the service
   369  // process. The service therefore reaches Running and honours Start/Stop just
   370  // like a production pg_timetable service.
   371  //
   372  // Opt-in and admin-only, like TestServiceLifecycle.
   373  func TestServiceRealStartStop(t *testing.T) {
   374  	if os.Getenv("PGTT_TEST_SERVICES") == "" {
   375  		t.Skip("skipping test that modifies system services: set PGTT_TEST_SERVICES=1 to enable")
   376  	}
   377  	requireSCM(t)
   378  
   379  	const name = "pg_timetable_realsvc_test"
   380  	exePath, err := os.Executable()
   381  	require.NoError(t, err)
   382  
   383  	// Install a minimal service. The persisted arguments are irrelevant here
   384  	// because TestMain short-circuits before the daemon ever connects to a DB.
   385  	cmdOpts := config.NewCmdOptions("--service=install", "--service-name="+name)
   386  	_ = serviceUninstall(name)
   387  	require.Equal(t, ExitCodeOK, serviceInstall(name, cmdOpts, exePath), "install must succeed")
   388  	t.Cleanup(func() { _ = serviceUninstall(name) })
   389  
   390  	m, err := connectManager()
   391  	require.NoError(t, err)
   392  	defer func() { _ = m.Disconnect() }()
   393  	s, err := m.OpenService(name)
   394  	require.NoError(t, err)
   395  	defer s.Close()
   396  
   397  	// Start the real service and wait until the SCM reports it Running.
   398  	require.Equal(t, ExitCodeOK, serviceControl(name, "start"), "start must succeed")
   399  	require.Eventually(t, func() bool {
   400  		st, qerr := s.Query()
   401  		return qerr == nil && st.State == svc.Running
   402  	}, 15*time.Second, servicePollingInterval, "service must reach Running")
   403  
   404  	// Stop it through the real entry point; stopService waits for Stopped.
   405  	require.Equal(t, ExitCodeOK, serviceControl(name, "stop"), "stop must succeed")
   406  	st, err := s.Query()
   407  	require.NoError(t, err)
   408  	assert.Equal(t, svc.Stopped, st.State, "service must be stopped")
   409  
   410  	require.Equal(t, ExitCodeOK, serviceUninstall(name), "uninstall must succeed")
   411  }

   413  // TestMain intercepts the case where the Service Control Manager launched this
   414  // test binary as a Windows service (see TestServiceRealStartStop). In that
   415  // process the package init() has already called svc.Run(winServiceRunner{}),
   416  // which drives the service. We provide a no-op cancel function so a graceful
   417  // Stop returns immediately, then wait for the service dispatcher to finish and
   418  // exit the process. Exiting is essential: it releases the .test.exe file so
   419  // the Go toolchain can delete it, and it prevents the normal test suite from
   420  // running inside the service process. Any other invocation runs the tests as
   421  // usual.
   422  func TestMain(m *testing.M) {
   423  	if isService, _ := svc.IsWindowsService(); isService {
   424  		setCancelFn(func() {})
   425  		<-serviceDone // wait until the SCM has stopped the service
   426  		os.Exit(0)
   427  	}
   428  	os.Exit(m.Run())
   429  }
