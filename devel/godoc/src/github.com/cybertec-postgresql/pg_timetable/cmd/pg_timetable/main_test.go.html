    23  // requireDocker skips the current test unless a Docker daemon capable of
    24  // running the Linux test containers is reachable through the same client
    25  // testcontainers uses (honoring DOCKER_HOST, docker:// / tcp:// / npipe://
    26  // endpoints and TESTCONTAINERS_* settings).
    27  //
    28  // The check is capability-based rather than OS-based: it queries the daemon's
    29  // OSType and only skips when it is not "linux". A Windows developer running
    30  // Docker Desktop with the WSL2/Linux backend therefore runs the full suite,
    31  // while a Windows-container-only engine (as on the windows-latest CI runner)
    32  // is skipped and covered by the Linux job instead.
    33  func requireDocker(t *testing.T) {
    34  	t.Helper()
    35  	ctx, cancel := context.WithTimeout(context.Background(), 10*time.Second)
    36  	defer cancel()
    37  	provider, err := testcontainers.NewDockerProvider()
    38  	if err != nil {
    39  		t.Skipf("no Docker provider available; skipping testcontainer-based test: %v", err)
    40  	}
    41  	defer func() { _ = provider.Close() }()
    42  	if err := provider.Health(ctx); err != nil {
    43  		t.Skipf("Docker daemon not reachable; skipping testcontainer-based test: %v", err)
    44  	}
    45  	info, err := provider.Client().Info(ctx, client.InfoOptions{})
    46  	if err != nil {
    47  		t.Skipf("cannot query Docker daemon info; skipping testcontainer-based test: %v", err)
    48  	}
    49  	if info.Info.OSType != "linux" {
    50  		t.Skipf("Docker daemon runs %q containers, not linux; skipping testcontainer-based test", info.Info.OSType)
    51  	}
    52  }

    59  // setupTestContainer starts a bare PostgreSQL container and returns the
    60  // connection string along with a cleanup function. Unlike the shared
    61  // testutils helper, it does NOT initialise the pg_timetable schema so that
    62  // run() can perform that step itself.
    63  func setupTestContainer(t *testing.T) (connStr string, cleanup func()) {
    64  	t.Helper()
    65  	requireDocker(t)
    66  	ctx := context.Background()
    67  	c, err := postgres.Run(
    68  		ctx,
    69  		"postgres:18-alpine",
    70  		postgres.WithDatabase("timetable"),
    71  		postgres.WithUsername("scheduler"),
    72  		postgres.WithPassword("somestrong"),
    73  		testcontainers.WithWaitStrategyAndDeadline(
    74  			60*time.Second,
    75  			wait.ForLog("database system is ready to accept connections").WithOccurrence(2),
    76  		),
    77  	)
    78  	require.NoError(t, err, "Failed to start PostgreSQL container")
    79  	cs, err := c.ConnectionString(ctx, "sslmode=disable")
    80  	if err != nil {
    81  		_ = c.Terminate(ctx)
    82  		t.Fatalf("Failed to get connection string: %v", err)
    83  	}
    84  	return cs, func() { _ = c.Terminate(ctx) }
    85  }

   124  // TestSetupCloseHandler verifies that sending SIGTERM causes the provided
   125  // cancel function to be called. Skipped on Windows where signal delivery to
   126  // the current process works differently.
   127  func TestSetupCloseHandler(t *testing.T) {
   128  	if runtime.GOOS == "windows" {
   129  		t.Skip("SIGTERM delivery to self is not supported on Windows")
   130  	}
   131  
   132  	ctx, cancel := context.WithCancel(context.Background())
   133  	done := make(chan struct{})
   134  	SetupCloseHandler(func() {
   135  		cancel()
   136  		close(done)
   137  	})
   138  
   139  	p, err := os.FindProcess(os.Getpid())
   140  	require.NoError(t, err)
   141  	require.NoError(t, p.Signal(syscall.SIGTERM))
   142  
   143  	select {
   144  	case <-done:
   145  	case <-time.After(3 * time.Second):
   146  		t.Fatal("cancel was not called within 3 s of receiving SIGTERM")
   147  	}
   148  	assert.ErrorIs(t, ctx.Err(), context.Canceled)
   149  }
