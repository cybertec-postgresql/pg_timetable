   103  func Parse(writer io.Writer) (*flags.Parser, error) {
   104  	cmdOpts := new(CmdOptions)
   105  	parser := flags.NewParser(cmdOpts, flags.PrintErrors)
   106  	var err error
   107  	if nonOptionArgs, err = parser.Parse(); err != nil {
   108  		if !flags.WroteHelp(err) && !cmdOpts.NoHelpMessage {
   109  			parser.WriteHelp(writer)
   110  			return nil, err
   111  		}
   112  	}
   113  	for _, f := range cmdOpts.Start.File {
   114  		if _, err := os.Stat(f); os.IsNotExist(err) {
   115  			return nil, err
   116  		}
   117  	}
   118  	//non-option arguments
   119  	if len(nonOptionArgs) > 0 && cmdOpts.ConnStr == "" {
   120  		cmdOpts.ConnStr = nonOptionArgs[0]
   121  	}
   122  	return parser, nil
   123  }
