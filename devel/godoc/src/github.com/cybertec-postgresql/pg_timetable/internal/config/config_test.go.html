    84  		{
    85  			name:    "sample ratio too high",
    86  			opts:    OTelOpts{SampleRatio: 1.5, MetricPeriod: 30, ShutdownTimeout: 5},
    87  			wantErr: "otel-sample-ratio must be between 0.0 and 1.0",
    88  		},
    89  		{
    90  			name:    "sample ratio negative",
    91  			opts:    OTelOpts{SampleRatio: -0.1, MetricPeriod: 30, ShutdownTimeout: 5},
    92  			wantErr: "otel-sample-ratio must be between 0.0 and 1.0",
    93  		},
    94  		{
    95  			name:    "metric period zero",
    96  			opts:    OTelOpts{SampleRatio: 1.0, MetricPeriod: 0, ShutdownTimeout: 5},
    97  			wantErr: "otel-metric-period must be > 0",
    98  		},
    99  		{
   100  			name:    "shutdown timeout zero",
   101  			opts:    OTelOpts{SampleRatio: 1.0, MetricPeriod: 30, ShutdownTimeout: 0},
   102  			wantErr: "otel-shutdown-timeout must be > 0",
   103  		},
   104  		{
   105  			name:    "unsupported endpoint scheme",
   106  			opts:    OTelOpts{SampleRatio: 1.0, MetricPeriod: 30, ShutdownTimeout: 5, Endpoint: "ftp://localhost:4317"},
   107  			wantErr: "unsupported OTel endpoint scheme: ftp",
   108  		},
   109  		{
   110  			name: "valid grpc endpoint",
   111  			opts: OTelOpts{SampleRatio: 1.0, MetricPeriod: 30, ShutdownTimeout: 5, Endpoint: "grpc://localhost:4317"},
   112  		},
   113  		{
   114  			name: "valid https endpoint",
   115  			opts: OTelOpts{SampleRatio: 1.0, MetricPeriod: 30, ShutdownTimeout: 5, Endpoint: "https://api.honeycomb.io"},
   116  		},
   117  	}
   118  	for _, tt := range tests {
   119  		t.Run(tt.name, func(t *testing.T) {
   120  			err := ValidateOTel(tt.opts)
   121  			if tt.wantErr == "" {
   122  				assert.NoError(t, err)
   123  			} else {
   124  				assert.EqualError(t, err, tt.wantErr)
   125  			}
   126  		})
