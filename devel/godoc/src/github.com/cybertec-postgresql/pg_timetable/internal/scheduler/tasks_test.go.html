    88  // TestSendMailResolvesSecret stores a secret for the running client, calls
    89  // taskSendMail (the actual code path executed by the scheduler) with a
    90  // reference, and asserts that the *decrypted* plaintext — not the literal
    91  // ${secret:...} reference — is what reaches the tasks.SendMail boundary.
    92  // tasks.NewDialer is overridden to capture the password argument instead of
    93  // performing a real SMTP dial, so a regression where taskSendMail stops
    94  // calling the resolver (and passes the unresolved reference through) is
    95  // caught here, unlike a test that exercises ResolveSecretsJSON directly.
    96  func TestSendMailResolvesSecret(t *testing.T) {
    97  	container, cleanup := testutils.SetupPostgresContainer(t)
    98  	defer cleanup()
    99  	pge := container.Engine
   100  	ctx := context.Background()
   101  	installPgcrypto(ctx, t, pge)
   102  
   103  	const name = "sendmail_resolve"
   104  	const pw = "real-secret-pw"
   105  	_, err := pge.ConfigDb.Exec(ctx,
   106  		`INSERT INTO timetable.secret (client_name, secret_name, value_enc)
   107  		 VALUES ($1, $2, pgp_sym_encrypt($3, $4))
   108  		 ON CONFLICT (client_name, secret_name) DO UPDATE SET value_enc = EXCLUDED.value_enc`,
   109  		pge.ClientName, name, pw, pge.SecretEncryptionKey)
   110  	require.NoError(t, err)
   111  
   112  	sch := New(pge, log.Init(config.LoggingOpts{LogLevel: "panic", LogDBLevel: "none"}), otel.NewNoop())
   113  
   114  	var gotPassword string
   115  	origNewDialer := tasks.NewDialer
   116  	tasks.NewDialer = func(_ string, _ int, _, password string) tasks.Dialer {
   117  		gotPassword = password
   118  		return fakeDialer{}
   119  	}
   120  	defer func() { tasks.NewDialer = origNewDialer }()
   121  
   122  	param := `{"serverhost":"127.0.0.1","serverport":1,"username":"u","senderaddr":"u@x","toaddr":["u@x"],"subject":"s","msgbody":"b","password":"${secret:` + name + `}"}`
   123  	_, err = taskSendMail(ctx, sch, param)
   124  	require.NoError(t, err)
   125  	assert.Equal(t, pw, gotPassword, "taskSendMail must pass the decrypted secret, not the literal reference, to SendMail")
   126  }
