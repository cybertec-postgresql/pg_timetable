    17  func TestExecuteChainCancelledContextCleansUp(t *testing.T) {
    18  	container, cleanup := testutils.SetupPostgresContainer(t)
    19  	defer cleanup()
    20  
    21  	pge := container.Engine
    22  	ctx := context.Background()
    23  	sch := New(pge, log.Init(config.LoggingOpts{LogLevel: "panic", LogDBLevel: "none"}), otel.NewNoop())
    24  
    25  	var chainID int
    26  	err := pge.ConfigDb.QueryRow(ctx, `
    27  		INSERT INTO timetable.chain(chain_name, max_instances)
    28  		VALUES ('cancel_cleanup_test', 16) RETURNING chain_id`).Scan(&chainID)
    29  	require.NoError(t, err)
    30  
    31  	_, err = pge.ConfigDb.Exec(ctx, `
    32  		INSERT INTO timetable.task(chain_id, task_order, kind, command)
    33  		VALUES ($1, 1, 'SQL', 'SELECT pg_sleep(0.5)')`, chainID)
    34  	require.NoError(t, err)
    35  
    36  	// Cancel the worker context while the task is sleeping, mimicking notify_chain_stop().
    37  	execCtx, cancel := context.WithCancel(ctx)
    38  	go func() {
    39  		time.Sleep(100 * time.Millisecond)
    40  		cancel()
    41  	}()
    42  
    43  	chains := make(chan Chain, 1)
    44  	chains <- Chain{ChainID: chainID, MaxInstances: 16}
    45  	sch.chainWorker(execCtx, chains)
    46  
    47  	// The stale active_chain row must be gone: cleanup must succeed despite the
    48  	// cancelled context.
    49  	var count int
    50  	err = pge.ConfigDb.QueryRow(ctx,
    51  		"SELECT count(*) FROM timetable.active_chain WHERE chain_id = $1", chainID).Scan(&count)
    52  	require.NoError(t, err)
    53  	assert.Zero(t, count, "active_chain entry must be removed even when the execution context was cancelled")
    54  }

    57  	container, cleanup := testutils.SetupPostgresContainer(t)
    58  	defer cleanup()
    59  
    60  	// A chain that runs longer than DetachedTimeout must still commit:
    61  	// the timeout bounds cleanup, not the chain.
    62  	defer func(d time.Duration) { log.DetachedTimeout = d }(log.DetachedTimeout)
    63  	log.DetachedTimeout = 200 * time.Millisecond
    64  
    65  	pge := container.Engine
    66  	ctx := context.Background()
    67  	sch := New(pge, log.Init(config.LoggingOpts{LogLevel: "panic", LogDBLevel: "none"}), otel.NewNoop())
    68  
    69  	_, err := pge.ConfigDb.Exec(ctx, "CREATE TABLE detached_timeout_test (id int)")
    70  	require.NoError(t, err)
    71  
    72  	var chainID int
    73  	err = pge.ConfigDb.QueryRow(ctx, `
    74  		INSERT INTO timetable.chain(chain_name, max_instances)
    75  		VALUES ('detached_timeout_test', 16) RETURNING chain_id`).Scan(&chainID)
    76  	require.NoError(t, err)
    77  
    78  	_, err = pge.ConfigDb.Exec(ctx, `
    79  		INSERT INTO timetable.task(chain_id, task_order, kind, command, autonomous)
    80  		VALUES ($1, 1, 'SQL', 'INSERT INTO detached_timeout_test SELECT 1 FROM pg_sleep(0.5)', false)`, chainID)
    81  	require.NoError(t, err)
    82  
    83  	sch.executeChain(ctx, Chain{ChainID: chainID, MaxInstances: 16})
    84  
    85  	var count int
    86  	err = pge.ConfigDb.QueryRow(ctx, "SELECT count(*) FROM detached_timeout_test").Scan(&count)
    87  	require.NoError(t, err)
    88  	assert.Equal(t, 1, count, "chain transaction must be committed")
