    79  func (pgxlogger *PgxLogger) Log(ctx context.Context, level tracelog.LogLevel, msg string, data map[string]any) {
    80  	logger := GetLogger(ctx)
    81  	if logger == FallbackLogger { //switch from standard to specified
    82  		logger = pgxlogger.l
    83  	}
    84  	if data != nil && noQueryArgs(ctx) {
    85  		delete(data, "args")
    86  	}
    87  	if data != nil {
    88  		logger = logger.WithFields(data)
    89  	}
    90  	switch level {
    91  	case tracelog.LogLevelTrace:
    92  		logger.WithField("PGX_LOG_LEVEL", level).Debug(msg)
    93  	case tracelog.LogLevelDebug, tracelog.LogLevelInfo: //pgx is way too chatty on INFO level
    94  		logger.Debug(msg)
    95  	case tracelog.LogLevelWarn:
    96  		logger.Warn(msg)
    97  	case tracelog.LogLevelError:
    98  		logger.Error(msg)
    99  	default:
   100  		logger.WithField("INVALID_PGX_LOG_LEVEL", level).Error(msg)
   101  	}
   102  }

   104  // WithoutQueryArgs marks ctx so that PgxLogger.Log drops the "args" field

   126  // DetachedContext returns a context that survives cancellation of ctx but keeps its logger,
