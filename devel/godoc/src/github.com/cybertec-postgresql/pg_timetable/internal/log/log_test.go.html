    46  func TestPgxLoggerDropsQueryArgs(t *testing.T) {
    47  	var buf bytes.Buffer
    48  	base := logrus.New()
    49  	base.SetOutput(&buf)
    50  	base.SetLevel(logrus.DebugLevel)
    51  	l := log.Init(config.LoggingOpts{LogLevel: "debug"})
    52  	_ = l
    53  	pgxl := log.NewPgxLogger(base)
    54  
    55  	ctx := context.Background()
    56  	pgxl.Log(log.WithLogger(ctx, base), tracelog.LogLevelDebug,
    57  		"Query", map[string]any{"sql": "SELECT $1", "args": []any{"secret-value"}})
    58  	assert.Contains(t, buf.String(), "SELECT $1")
    59  	assert.Contains(t, buf.String(), "secret-value", "args must be logged when context is unmarked")
    60  
    61  	buf.Reset()
    62  	pgxl.Log(log.WithLogger(log.WithoutQueryArgs(ctx), base), tracelog.LogLevelDebug,
    63  		"Query", map[string]any{"sql": "SELECT $1", "args": []any{"secret-value"}})
    64  	assert.Contains(t, buf.String(), "SELECT $1", "sql must remain under WithoutQueryArgs")
    65  	assert.NotContains(t, buf.String(), "secret-value", "args must be dropped under WithoutQueryArgs")
    66  }
