    71  func (pge *PgEngine) ExecLocalSQLTask(ctx context.Context, tx pgx.Tx, task *ChainTask, paramValues []string) (err error) {
    72  	if err := pge.SetRole(ctx, tx, task.RunAs); err != nil {
    73  		return err
    74  	}
    75  	if task.IgnoreError {
    76  		pge.MustSavepoint(ctx, tx, task.TaskID)
    77  	}
    78  	pge.SetCurrentTaskContext(ctx, tx, task.ChainID, task.TaskID)
    79  	err = pge.ExecuteSQLCommand(ctx, tx, task, paramValues)
    80  	if err != nil && task.IgnoreError {
    81  		pge.MustRollbackToSavepoint(ctx, tx, task.TaskID)
    82  	}
    83  	if task.RunAs > "" {
    84  		pge.ResetRole(ctx, tx)
    85  	}
    86  	return
    87  }

   130  // ExecuteSQLCommand executes chain command with parameters inside transaction.
   131  //
   132  // ${secret:name} references inside each parameter are resolved *before*
   133  // unmarshalling into the bound args, while the original (unresolved) `val`
   134  // is the only string passed to LogTaskExecution. When resolution substitutes
   135  // at least one secret, the bound-argument query is issued under a context
   136  // marked with log.WithoutQueryArgs so the pgx tracer does not persist
   137  // resolved values to timetable.log.
   138  func (pge *PgEngine) ExecuteSQLCommand(ctx context.Context, executor executor, task *ChainTask, paramValues []string) (err error) {
   139  	var params []any
   140  	var errCodes = map[bool]int{false: 0, true: -1}
   141  	if strings.TrimSpace(task.Command) == "" {
   142  		return errors.New("SQL command cannot be empty")
   143  	}
   144  	if len(paramValues) == 0 { //mimic empty param
   145  		ct, e := executor.Exec(ctx, task.Command)
   146  		pge.LogTaskExecution(context.Background(), task, errCodes[e != nil], ct.String(), "")
   147  		return e
   148  	}
   149  	for _, val := range paramValues {
   150  		if val == "" {
   151  			continue
   152  		}
   153  		task.StartedAt = time.Now() // reset start time for each parameter set execution
   154  		resolved, names, rerr := pge.ResolveSecretsJSON(ctx, val)
   155  		if rerr != nil {
   156  			return rerr
   157  		}
   158  		if parseErr := json.Unmarshal([]byte(resolved), &params); parseErr != nil {
   159  			// Use the original (unresolved) val here: resolved may contain
   160  			// decrypted secret plaintext, which must never reach an error
   161  			// message that gets logged.
   162  			err = errors.Join(err, fmt.Errorf("failed to parse parameter %s: %w", val, parseErr))
   163  			return
   164  		}
   165  		execCtx := ctx
   166  		if len(names) > 0 {
   167  			execCtx = log.WithoutQueryArgs(ctx)
   168  		}
   169  		ct, e := executor.Exec(execCtx, task.Command, params...)
   170  		err = errors.Join(err, e)
   171  		pge.LogTaskExecution(context.Background(), task, errCodes[e != nil], ct.String(), val)
   172  	}
   173  	return
   174  }
