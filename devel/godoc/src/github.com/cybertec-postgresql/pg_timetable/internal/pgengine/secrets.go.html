    26  // resolveRefs is the shared engine used by both ResolveSecretsJSON and
    27  // ResolveSecretsConnString.
    28  //
    29  // If s does not contain the literal substring `${secret:`, it is returned
    30  // byte-identical with no parsing, no regexp evaluation, and no database call.
    31  // Otherwise, each match is resolved exactly once against
    32  // timetable.resolve_secret using pge.ClientName as the fixed scope, and quote
    33  // is applied to the resolved value before substitution. Resolved values are
    34  // never re-scanned.
    35  func (pge *PgEngine) resolveRefs(
    36  	ctx context.Context, s string,
    37  	quote func(value string, m []int, in string) string,
    38  ) (string, []string, error) {
    39  	if !strings.Contains(s, secretRefSubstring) {
    40  		return s, nil, nil
    41  	}
    42  	// Pre-flight: if any reference is present and the key is empty, fail fast
    43  	// with a descriptive error.
    44  	if pge.SecretEncryptionKey == "" {
    45  		refNames := extractRefNames(s)
    46  		return "", refNames, fmt.Errorf(
    47  			"secret references found (%s) but SecretEncryptionKey is not configured; set PGTT_SECRET_KEY/--secret-key",
    48  			strings.Join(refNames, ", "))
    49  	}
    50  	markedCtx := log.WithoutQueryArgs(ctx)
    51  	var (
    52  		out   strings.Builder
    53  		names []string
    54  		last  int
    55  	)
    56  	for _, m := range secretRefPattern.FindAllStringSubmatchIndex(s, -1) {
    57  		name := s[m[2]:m[3]]
    58  		var plaintext *string
    59  		err := pge.ConfigDb.QueryRow(markedCtx, resolveSecretSQL, name, pge.ClientName, pge.SecretEncryptionKey).Scan(&plaintext)
    60  		if err != nil {
    61  			// pgp_sym_decrypt raises "Wrong key or corrupt data".
    62  			// Surface it wrapped with the secret name; never as not-found.
    63  			if isWrongKey(err) {
    64  				return "", append(names, name), fmt.Errorf(
    65  					`secret %q: wrong key or corrupt data`, name)
    66  			}
    67  			// pgcrypto is absent. resolve_secret raises
    68  			// feature_not_supported (SQLSTATE 0A000). Wrap with
    69  			// the secret name and a statement that installing pgcrypto is the
    70  			// database administrator's responsibility. This is a per-task
    71  			// error only — never a startup failure.
    72  			if isMissingPgcrypto(err) {
    73  				return "", append(names, name), fmt.Errorf(
    74  					`secret %q: pgcrypto extension is not installed; `+
    75  						`installing it is the database administrator's responsibility`,
    76  					name)
    77  			}
    78  			return "", append(names, name), fmt.Errorf(
    79  				"secret %q: %w", name, err)
    80  		}
    81  		if plaintext == nil {
    82  			// Missing secret (one row containing NULL). Indistinguishable
    83  			// across client scopes.
    84  			return "", append(names, name), fmt.Errorf(
    85  				`secret %q not found for client %q`, name, pge.ClientName)
    86  		}
    87  		out.WriteString(s[last:m[0]])
    88  		out.WriteString(quote(*plaintext, m, s))
    89  		names = append(names, name)
    90  		last = m[1]
    91  	}
    92  	out.WriteString(s[last:])
    93  	return out.String(), uniqueRefNames(strings.Join(names, ",")), nil
    94  }

   115  func uniqueRefNames(s string) []string {
   116  	if s == "" {
   117  		return nil
   118  	}
   119  	parts := strings.Split(s, ",")
   120  	seen := make(map[string]struct{}, len(parts))
   121  	out := make([]string, 0, len(parts))
   122  	for _, p := range parts {
   123  		if _, ok := seen[p]; ok {
   124  			continue
   125  		}
   126  		seen[p] = struct{}{}
   127  		out = append(out, p)
   128  	}
   129  	return out
   130  }

   132  // extractRefNames extracts the unique ${secret:name} identifiers referenced
   133  // in s (which may be arbitrary JSON or conninfo text), unlike uniqueRefNames
   134  // which only dedupes an already-extracted, comma-joined list of names.
   135  func extractRefNames(s string) []string {
   136  	matches := secretRefPattern.FindAllStringSubmatch(s, -1)
   137  	names := make([]string, 0, len(matches))
   138  	for _, m := range matches {
   139  		if len(m) > 1 {
   140  			names = append(names, m[1])
   141  		}
   142  	}
   143  	return uniqueRefNames(strings.Join(names, ","))

   146  // ResolveSecretsJSON resolves ${secret:name} references inside the string
   147  // leaves of a jsonb-encoded parameter value and returns the re-encoded JSON.
   148  // Resolved values are never re-scanned.
   149  //
   150  // If s does not contain the literal substring "${secret:" it is returned
   151  // byte-identical with no parsing, no database call.
   152  func (pge *PgEngine) ResolveSecretsJSON(ctx context.Context, s string) (resolved string, names []string, err error) {
   153  	if !strings.Contains(s, secretRefSubstring) {
   154  		return s, nil, nil
   155  	}
   156  	if pge.SecretEncryptionKey == "" {
   157  		return "", extractRefNames(s), fmt.Errorf(
   158  			"secret references found but SecretEncryptionKey is not configured; set PGTT_SECRET_KEY/--secret-key")
   159  	}
   160  	var doc any
   161  	if derr := json.Unmarshal([]byte(s), &doc); derr != nil {
   162  		return "", nil, fmt.Errorf("resolve json: %w", derr)
   163  	}
   164  	var (
   165  		allNames []string
   166  		walkErr  error
   167  	)
   168  	walkJSON(doc, func(v *string) {
   169  		if v == nil {
   170  			return
   171  		}
   172  		// resolveRefs short-circuits on no substring; safe to call per leaf.
   173  		sub, names, rerr := pge.resolveRefs(ctx, *v, quoteJSONLeaf)
   174  		if rerr != nil {
   175  			walkErr = errors.Join(walkErr, rerr)
   176  			return
   177  		}
   178  		*v = sub
   179  		allNames = append(allNames, names...)
   180  	})
   181  	if walkErr != nil {
   182  		return "", uniqueRefNames(strings.Join(allNames, ",")), walkErr
   183  	}
   184  	out, merr := json.Marshal(doc)
   185  	if merr != nil {
   186  		return "", uniqueRefNames(strings.Join(allNames, ",")), fmt.Errorf("encode resolved json: %w", merr)
   187  	}
   188  	return string(out), uniqueRefNames(strings.Join(allNames, ",")), nil
   189  }

   198  // walkJSON invokes fn on every JSON string leaf reachable from v. v is
   199  // mutated in place.
   200  func walkJSON(v any, fn func(*string)) {
   201  	switch x := v.(type) {
   202  	case map[string]any:
   203  		for k, child := range x {
   204  			if s, ok := child.(string); ok {
   205  				if strings.Contains(s, secretRefSubstring) {
   206  					p := s
   207  					fn(&p)
   208  					x[k] = p
   209  				}
   210  			} else {
   211  				walkJSON(child, fn)
   212  			}
   213  		}
   214  	case []any:
   215  		for i, child := range x {
   216  			if s, ok := child.(string); ok {
   217  				if strings.Contains(s, secretRefSubstring) {
   218  					p := s
   219  					fn(&p)
   220  					x[i] = p
   221  				}
   222  			} else {
   223  				walkJSON(child, fn)
   224  			}
   225  		}
   226  	}
   227  }

   238  // quoteConnInfoValue applies libpq conninfo quoting to a resolved secret
   239  // value. If the reference in the template is already delimited by single
   240  // quotes (e.g. `password='${secret:pw}'`), the wrapping is omitted and only
   241  // `\` and `'` are escaped, so the existing delimiters are not doubled.
   242  // An empty value is emitted as `”` because a bare `password=`
   243  // followed by whitespace would swallow the next token.
   244  func quoteConnInfoValue(value string, in string, m []int) string {
   245  	if value == "" {
   246  		return "''"
   247  	}
   248  	needsQuoting := strings.ContainsAny(value, " \t\n'\\")
   249  	if !needsQuoting {
   250  		return value
   251  	}
   252  	if isAlreadySingleQuoted(in, m) {
   253  		return escapeConnInfo(value)
   254  	}
   255  	return "'" + escapeConnInfo(value) + "'"
   256  }

   258  func escapeConnInfo(v string) string {
   259  	r := strings.NewReplacer(`\`, `\\`, `'`, `\'`)
   260  	return r.Replace(v)
   261  }

   263  // isAlreadySingleQuoted returns true when the bytes immediately surrounding
   264  // the reference form a `password='…'` style template.
   265  func isAlreadySingleQuoted(in string, m []int) bool {
   266  	if m[0] == 0 || m[1] >= len(in) {
   267  		return false
   268  	}
   269  	return in[m[0]-1] == '\'' && in[m[1]] == '\''
   270  }

   272  // CheckSecretConfig logs an error when timetable.secret contains rows but no
   273  // encryption key is configured. It performs no query when SecretEncryptionKey
   274  // is non-empty. A failure of the check itself is logged, never fatal.
   275  func (pge *PgEngine) CheckSecretConfig(ctx context.Context) error {
   276  	if pge.SecretEncryptionKey != "" {
   277  		return nil
   278  	}
   279  	markedCtx := log.WithoutQueryArgs(ctx)
   280  	var count int64
   281  	if err := pge.ConfigDb.QueryRow(markedCtx, `SELECT timetable.secret_count()`).Scan(&count); err != nil {
   282  		pge.l.WithError(err).Error("Cannot check timetable.secret configuration")
   283  		return nil
   284  	}
   285  	if count > 0 {
   286  		pge.l.WithField("secret_count", count).Error(
   287  			"timetable.secret contains rows but SecretEncryptionKey is not configured; set PGTT_SECRET_KEY/--secret-key")
   288  	}
   289  	return nil
   290  }
