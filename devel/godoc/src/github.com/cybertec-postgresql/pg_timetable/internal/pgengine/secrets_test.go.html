   110  // assertNoExtensionDMLInDDL walks every SQL file under internal/pgengine/sql/
   111  // and fails the test if any file contains CREATE EXTENSION or ALTER EXTENSION.
   112  // Walked from the test working directory; resolves the module root by walking
   113  // upward until go.mod is found.
   114  func assertNoExtensionDMLInDDL(t *testing.T) {
   115  	t.Helper()
   116  	root, err := findModuleRoot()
   117  	require.NoError(t, err)
   118  	dir := filepath.Join(root, "internal", "pgengine", "sql")
   119  	err = filepath.WalkDir(dir, func(path string, d fs.DirEntry, walkErr error) error {
   120  		if walkErr != nil {
   121  			return walkErr
   122  		}
   123  		if d.IsDir() {
   124  			return nil
   125  		}
   126  		if !strings.HasSuffix(path, ".sql") {
   127  			return nil
   128  		}
   129  		b, rerr := os.ReadFile(path)
   130  		if rerr != nil {
   131  			return rerr
   132  		}
   133  		s := string(b)
   134  		// Reject actual statements, but allow the substrings to appear inside
   135  		// SQL string literals or comments if and only if they are escaped /
   136  		// commented out. The simplest correct check is: no top-level
   137  		// `CREATE EXTENSION` or `ALTER EXTENSION` statement — i.e. a line
   138  		// beginning with either keyword, ignoring leading whitespace.
   139  		for _, line := range strings.Split(s, "\n") {
   140  			trimmed := strings.TrimSpace(line)
   141  			upper := strings.ToUpper(trimmed)
   142  			if strings.HasPrefix(upper, "CREATE EXTENSION") ||
   143  				strings.HasPrefix(upper, "ALTER EXTENSION") {
   144  				t.Errorf("forbidden extension DML in %s: %s", path, trimmed)
   145  			}
   146  		}
   147  		return nil
   148  	})
   149  	require.NoError(t, err)
   150  }

   152  func findModuleRoot() (string, error) {
   153  	cwd, err := os.Getwd()
   154  	if err != nil {
   155  		return "", err
   156  	}
   157  	dir := cwd
   158  	for {
   159  		if _, err := os.Stat(filepath.Join(dir, "go.mod")); err == nil {
   160  			return dir, nil
   161  		}
   162  		parent := filepath.Dir(dir)
   163  		if parent == dir {
   164  			return "", errors.New("go.mod not found")
   165  		}
   166  		dir = parent
   167  	}

   587  // TestPgxTracerRedactsSecretArgs: the pgx tracer in this codebase is
   588  // `log.NewPgxLogger`, wired via bootstrap.getPgxConnConfig. When the resolver
   589  // calls `timetable.resolve_secret` under a context marked with
   590  // `log.WithoutQueryArgs`, PgxLogger.Log MUST drop the `args` field (which
   591  // carries the encryption key as a bound parameter) while retaining `sql`.
   592  // This test drives PgxLogger directly so the assertion is independent of
   593  // whether the testcontainer's log level is high enough to persist tracer
   594  // output to `timetable.log`.
   595  func TestPgxTracerRedactsSecretArgs(t *testing.T) {
   596  	// Unmarked context: args + sql must both appear.
   597  	unmarkedBuf := &captureBuf{}
   598  	unmarkedL := newLogrusInto(unmarkedBuf)
   599  	plUnmarked := log.NewPgxLogger(unmarkedL)
   600  	plUnmarked.Log(context.Background(),
   601  		pgxLogLevel("Debug"),
   602  		"Query",
   603  		map[string]any{"sql": "SELECT 1", "args": []any{"k", "v"}},
   604  	)
   605  	assert.Contains(t, unmarkedBuf.String(), "SELECT 1",
   606  		"sql must always be logged verbatim")
   607  	assert.Contains(t, unmarkedBuf.String(), "k",
   608  		"unmarked context: args are retained")
   609  
   610  	// Marked context: args must NOT appear, sql MUST.
   611  	markedBuf := &captureBuf{}
   612  	markedL := newLogrusInto(markedBuf)
   613  	plMarked := log.NewPgxLogger(markedL)
   614  	ctx := log.WithoutQueryArgs(context.Background())
   615  	plMarked.Log(ctx,
   616  		pgxLogLevel("Debug"),
   617  		"Query",
   618  		map[string]any{
   619  			"sql":  "SELECT timetable.resolve_secret($1, $2, $3)",
   620  			"args": []any{"tracer_redact", "marker-pw", "marker-key"},
   621  		},
   622  	)
   623  	out := markedBuf.String()
   624  	assert.Contains(t, out, "SELECT timetable.resolve_secret",
   625  		"sql must be retained")
   626  	assert.NotContains(t, out, "marker-pw",
   627  		"plaintext must not leak through args under WithoutQueryArgs")
   628  	assert.NotContains(t, out, "marker-key",
   629  		"encryption key must not leak through args under WithoutQueryArgs")
   630  	assert.NotContains(t, out, "tracer_redact",
   631  		"secret name (an arg) must not leak through args under WithoutQueryArgs")
   632  }
